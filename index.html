return; } const allPts = [pStart, ...waypoints, endPoint]; const engine = lg('routerEngine') || 'osrm'; const orsKey = lg('orsKey') || ''; clearRoutePolylines(); routesData = []; let rawRoutes = []; if (engine === 'ors' && orsKey) { const bodyQuery = { coordinates: allPts.map(p => [p[1], p[0]]), preference: "recommended", alternative_routes: { target_count: 3 } }; if (isScooterMode249) bodyQuery.options = { avoid_features: ["highways", "tollways", "ferries"] }; try { const res = await fetch(`https://api.openrouteservice.org/v2/directions/driving-car/geojson`, { method: 'POST', headers: { 'Authorization': orsKey, 'Content-Type': 'application/json' }, body: JSON.stringify(bodyQuery) }); const data = await res.json(); if (data.features) { rawRoutes = data.features.map(f => ({ coords: f.geometry.coordinates.map(c => [c[1], c[0]]), distance: f.properties.summary.distance })); } } catch(e) {} } // Резервний OSRM if (rawRoutes.length === 0) { const coordStr = allPts.map(p => `${p[1]},${p[0]}`).join(';'); try { const res = await fetch(`https://router.project-osrm.org/route/v1/driving/${coordStr}?overview=full&geometries=geojson&alternatives=true`); const data = await res.json(); if (data.code === 'Ok' && data.routes) { rawRoutes = data.routes.map(r => ({ coords: r.geometry.coordinates.map(c => [c[1], c[0]]), distance: r.distance })); } } catch (e) {} } if (rawRoutes.length > 0) { // Видаляємо дублікати та занадто подібні маршрути (різниця < 50м) routesData = filterUniqueRoutes(rawRoutes); activeRouteIndex = 0; drawRoutesOnMap(); updateHudUI(); } else { alert("Не вдалося побудувати маршрут."); } } // Алгоритм відсіювання дубльованих маршрутів function filterUniqueRoutes(routes) { const unique = []; routes.forEach(rt => { const isDuplicate = unique.some(u => { const distDiff = Math.abs(u.distance - rt.distance); // Якщо дистанції збігаються майже до метрів (менше 50м різниці), вважаємо дублікатом return distDiff < 50; }); if (!isDuplicate) { unique.push(rt); } }); return unique; }
