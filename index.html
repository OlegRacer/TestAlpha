async function buildRoute(fromCurrentGps = false) { let start = userCoords; if (!fromCurrentGps) { const startVal = document.getElementById('startInput').value; if (customStartCoords && startVal.includes('(Карта)')) { start = customStartCoords; } else if (startVal.trim() !== '') { const g = await geocode(startVal); if (g) start = g; } } let dest = activeDestCoords; if (!fromCurrentGps || !dest) { const destInput = document.getElementById('destInput'); if (destInput.dataset.lat && destInput.value.includes('(Карта)')) { dest = [parseFloat(destInput.dataset.lat), parseFloat(destInput.dataset.lng)]; } else { dest = await geocode(destInput.value); } } if (!dest) { alert('Будь ласка, вкажіть фініш'); return; } activeDestCoords = dest; let routeWaypoints = [start]; const wpElements = document.querySelectorAll('.wp-input'); for (let el of wpElements) { if (el.dataset.coords) { routeWaypoints.push(JSON.parse(el.dataset.coords)); } else if (el.value.trim() !== '') { const g = await geocode(el.value); if (g) routeWaypoints.push(g); } } const isAvoidTolls = document.getElementById('avoidTolls').checked; const isShortest = document.getElementById('shortestRoute').checked; const isCurvy = document.getElementById('curvyMode').checked; const isScenic = document.getElementById('scenicMode').checked; const isBike = document.getElementById('bikeMode').checked; if (isCurvy && wpElements.length === 0) { routeWaypoints.splice(1, 0, ...generateCurvyWaypoints(start, dest)); } else if (isScenic && wpElements.length === 0) { routeWaypoints.splice(1, 0, ...generateScenicWaypoints(start, dest)); } routeWaypoints.push(dest); // ЛОГІКА ПРОФІЛЮ: // Якщо вибрано "Найкоротший шлях", беремо профіль 'bike', бо він шукає найкоротші зрізи по місту. // В іншому випадку беремо 'driving' (найшвидший автошлях). let osrmProfile = 'driving'; if (isBike || isShortest) { osrmProfile = 'bike'; } const coordsStr = routeWaypoints.map(c => `${c[1]},${c[0]}`).join(';'); // Запитуємо маршрут const url = `https://router.project-osrm.org/route/v1/${osrmProfile}/${coordsStr}?overview=full&geometries=geojson&steps=true&alternatives=true`; try { const res = await fetch(url); const data = await res.json(); if (data.routes && data.routes.length > 0) { // Обираємо з усіх повернених варіантів той, де найменша відстань let route = data.routes.reduce((min, r) => r.distance < min.distance ? r : min, data.routes[0]); if (routeLayer) map.removeLayer(routeLayer); let routeColor = '#2563EB'; if (isCurvy) routeColor = '#D97706'; if (isAvoidTolls) routeColor = '#059669'; if (isShortest) routeColor = '#9333EA'; // Фіолетовий колір для найкоротшого шляху routeLayer = L.geoJSON(route.geometry, { style: { color: routeColor, weight: 6, opacity: 0.9 } }).addTo(map); map.fitBounds(routeLayer.getBounds(), { padding: [40, 40] }); currentSteps = []; route.legs.forEach(leg => { currentSteps.push(...parseSteps(leg.steps)); }); nextStepIndex = 0; offRouteWarned = false; document.getElementById('recalculateBtn').style.display = 'none'; const distKm = (route.distance / 1000).toFixed(1); document.getElementById('distDisplay').innerText = `${distKm} км`; document.getElementById('startDriveBtn').style.display = 'block'; adjustControlsPosition(); if (fromCurrentGps) { VoiceGuidance.speak(`Маршрут оновлено. Продовжуйте рух.`); } else { VoiceGuidance.speak(`Маршрут розраховано. Натисніть Поїхали.`); } updateHUD(); } } catch (e) { alert('Помилка завантаження маршруту'); } }
